# Native project/filesystem module; configured after kasane_core and kasane_moc3.
find_package(PNG REQUIRED)
add_library(kasane_filesystem STATIC src/filesystem.cpp)
kasane_warnings(kasane_filesystem)
target_include_directories(kasane_filesystem PUBLIC include)
target_compile_features(kasane_filesystem PUBLIC cxx_std_20)
set_target_properties(kasane_filesystem PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(kasane_package STATIC src/package.cpp)
kasane_warnings(kasane_package)
target_link_libraries(kasane_package PUBLIC kasane_moc3 kasane_filesystem PRIVATE PNG::PNG)
set_target_properties(kasane_package PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(kasane_document STATIC src/project_codec.cpp src/project_resources.cpp src/project.cpp)
kasane_warnings(kasane_document)
target_include_directories(kasane_document PUBLIC include)
target_include_directories(kasane_document SYSTEM PRIVATE vendor)
target_link_libraries(kasane_document PUBLIC kasane_package PRIVATE PNG::PNG OpenSSL::Crypto)
set_target_properties(kasane_document PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(BUILD_TESTING)
  add_executable(kasane_document_tests tests/project_tests.cpp)
  kasane_warnings(kasane_document_tests)
  target_include_directories(kasane_document_tests SYSTEM PRIVATE vendor)
  target_compile_definitions(kasane_document_tests PRIVATE KASANE_SAMPLE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../samples/m2-complete")
  target_link_libraries(kasane_document_tests PRIVATE kasane_document PNG::PNG)
  add_test(NAME kasane_document_tests COMMAND kasane_document_tests)
endif()
if(BUILD_TESTING)
  add_executable(kasane_document_tool tests/project_tool.cpp)
  kasane_warnings(kasane_document_tool)
  target_include_directories(kasane_document_tool SYSTEM PRIVATE vendor)
  target_link_libraries(kasane_document_tool PRIVATE kasane_document)

  add_executable(kasane_cpp_baseline ${CMAKE_CURRENT_SOURCE_DIR}/../../benches/cpp_baseline/benchmark_main.cpp)
  kasane_warnings(kasane_cpp_baseline)
  target_include_directories(kasane_cpp_baseline SYSTEM PRIVATE vendor)
  target_link_libraries(kasane_cpp_baseline PRIVATE kasane_document)
endif()
