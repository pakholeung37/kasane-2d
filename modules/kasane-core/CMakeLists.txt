cmake_minimum_required(VERSION 3.21)
project(kasane_core LANGUAGES CXX)
option(KASANE_STRICT_WARNINGS "Treat warnings in owned native targets as errors" ON)
function(kasane_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
    if(KASANE_STRICT_WARNINGS)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    # Fixtures intentionally initialize only the fields relevant to each test.
    if(target MATCHES "_tests$")
      target_compile_options(${target} PRIVATE -Wno-missing-field-initializers)
    endif()
    if(KASANE_STRICT_WARNINGS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
endfunction()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/core_sources.txt")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/core_sources.txt" KASANE_CORE_SOURCES)
list(TRANSFORM KASANE_CORE_SOURCES PREPEND "${CMAKE_CURRENT_SOURCE_DIR}/src/")
add_library(kasane_core STATIC ${KASANE_CORE_SOURCES})
kasane_warnings(kasane_core)
target_include_directories(kasane_core PUBLIC include)
target_include_directories(kasane_core SYSTEM PRIVATE ../purism-core/include)
target_compile_features(kasane_core PUBLIC cxx_std_20)
set_target_properties(kasane_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(kasane_moc3 STATIC src/moc3.cpp)
kasane_warnings(kasane_moc3)
target_link_libraries(kasane_moc3 PUBLIC kasane_core)
set_target_properties(kasane_moc3 PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(KASANE_BUILD_PACKAGE "Build PNG-backed package publication" ON)
if(KASANE_BUILD_PACKAGE)
  find_package(PNG REQUIRED)
  add_library(kasane_package STATIC src/package.cpp)
  kasane_warnings(kasane_package)
  target_link_libraries(kasane_package PUBLIC kasane_moc3 PRIVATE PNG::PNG)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(kasane_core_tests tests/core_tests.cpp)
  kasane_warnings(kasane_core_tests)
  target_link_libraries(kasane_core_tests PRIVATE kasane_core)
  add_test(NAME kasane_core_tests COMMAND kasane_core_tests)
  add_executable(kasane_keyform_tests tests/keyform_tests.cpp)
  kasane_warnings(kasane_keyform_tests)
  target_link_libraries(kasane_keyform_tests PRIVATE kasane_core)
  add_test(NAME kasane_keyform_tests COMMAND kasane_keyform_tests)
  add_executable(kasane_deformer_tests tests/deformer_tests.cpp)
  kasane_warnings(kasane_deformer_tests)
  target_link_libraries(kasane_deformer_tests PRIVATE kasane_core)
  add_test(NAME kasane_deformer_tests COMMAND kasane_deformer_tests)
endif()

# Optional compatibility harness. The core/codec themselves have no Core ABI
# dependency. Each provider runs in a separate process to avoid symbol mixing.
option(KASANE_MOC3_CONFORMANCE "Build MOC3 Core compatibility tests" ON)
set(KASANE_CUBISM_ROOT "" CACHE PATH "Local Cubism Native SDK root")
if(BUILD_TESTING AND KASANE_MOC3_CONFORMANCE)
  if(NOT KASANE_BUILD_PACKAGE)
    message(FATAL_ERROR "MOC3 conformance requires KASANE_BUILD_PACKAGE=ON")
  endif()
  enable_language(C)
  if(NOT TARGET PurismCore)
    add_subdirectory(../purism-core ${CMAKE_CURRENT_BINARY_DIR}/purism)
  endif()
  if(KASANE_STRICT_WARNINGS)
    if(MSVC)
      target_compile_options(PurismCore PRIVATE /WX)
    else()
      target_compile_options(PurismCore PRIVATE -Werror)
    endif()
  endif()
  if(TARGET stageplay)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME validation_negative_controls
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/test_validation_contract.py
              $<TARGET_FILE:stageplay>)
  endif()
  if(NOT WIN32)
    add_test(NAME purism_c99_bundle
      COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/check_purism_bundle.sh
              ${CMAKE_C_COMPILER}
              ${CMAKE_CURRENT_SOURCE_DIR}/../purism-core/scripts/bundle.sh)
  endif()
  add_executable(kasane_moc3_purism_tests tests/moc3_tests.cpp)
  kasane_warnings(kasane_moc3_purism_tests)
  target_link_libraries(kasane_moc3_purism_tests PRIVATE kasane_package PurismCore PNG::PNG)
  add_test(NAME kasane_moc3_purism_tests COMMAND kasane_moc3_purism_tests)
  if(KASANE_CUBISM_ROOT)
    if(APPLE)
      set(_cubism_platform macos)
    elseif(UNIX)
      set(_cubism_platform linux)
    else()
      message(FATAL_ERROR "Set up an official Core library for this platform before conformance")
    endif()
    set(_cubism_library "${KASANE_CUBISM_ROOT}/Core/lib/${_cubism_platform}/${CMAKE_SYSTEM_PROCESSOR}/libLive2DCubismCore.a")
    if(NOT EXISTS "${_cubism_library}")
      message(FATAL_ERROR "Missing official Core: ${_cubism_library}")
    endif()
    add_executable(kasane_moc3_official_tests tests/moc3_tests.cpp)
    kasane_warnings(kasane_moc3_official_tests)
    target_compile_definitions(kasane_moc3_official_tests PRIVATE KASANE_OFFICIAL_CORE)
    target_include_directories(kasane_moc3_official_tests PRIVATE "${KASANE_CUBISM_ROOT}/Core/include")
    target_link_libraries(kasane_moc3_official_tests PRIVATE kasane_package PNG::PNG "${_cubism_library}")
    add_test(NAME kasane_moc3_official_tests COMMAND kasane_moc3_official_tests)
  endif()
endif()
